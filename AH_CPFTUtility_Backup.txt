    //THIS METHOD IS USED TO CREATE BUDGET RECORDS
    Public Static Set<Id> createBudgets(List<pse__Budget__c> budgetList)
    {
        Set<Id> budIds = new Set<Id>();
        if(!budgetList.isEmpty())
        {
            Database.SaveResult[] srList = Database.insert(budgetList);
            // Iterate through each returned result
            for (Database.SaveResult sr : srList)
            {
                if(sr.isSuccess())
                {
                    budIds.add(sr.getId());
                }
                else
                {
                    // Operation failed, so get all errors                
                    for(Database.Error err : sr.getErrors()) 
                    {
                        System.debug(err.getStatusCode() + ': ' + err.getMessage());
                        System.debug('Budget fields that affected this error: ' + err.getFields());
                    }
                }
            }
            
            //UPDATE ASSETS RECORDS WITH ADDED TO PROJECT CHECKBOX AS TRUE FROM NEWLY CREATED BUDGETS
            if(!budIds.isempty())
            {
                Set<Id> assetIds = new set<Id>();
                for(pse__Budget__c newBud : [Select Id, Asset_Line_Item__c from pse__Budget__c where Id IN :budIds])
                {
                    assetIds.add(newBud.Asset_Line_Item__c);    
                }
                
                if(!assetIds.isempty())
                {
                    updateAssets(assetIds);    
                }
            }
        }
        return budIds;
    }

    @Future
    Public Static Void createProjectTasks(String selectedProjId,String newlyCreatedProjId)
    {
        try
        {
                if(selectedProjId != null && newlyCreatedProjId != null)
                {
                    //QUERY PROJECT TASKS RECORD BASED ON SELECTED PROJECT ID & USE FIELD SET TO GET FIELDS FOR QUERY
                    String queryString = 'select id ';							// OBJECT NAME			FIELD SET NAME		
                    for(Schema.FieldSetMember fld : AH_CPFTUtility.readFieldSet('pse__Project_Task__c','AH_CreateProjectFromTemplateCopy'))
                    {
                        queryString += ', ' + fld.getFieldPath();
                    }
                    queryString += ' from pse__Project_Task__c where pse__Project__c =: selectedProjId';
                    
                    List<sObject> insertProjectTasks = new  List<sObject>();
                    Map<String,List<pse__Project_Task__c>> childProjectTasks = new  Map<String,List<pse__Project_Task__c>>();
                    for(pse__Project_Task__c pTask : Database.query(queryString))
                    {
                        if(pTask.pse__Parent_Task__c != null)
                        {
                            if(childProjectTasks.containskey(pTask.pse__Parent_Task__c))
                            {
                                childProjectTasks.get(pTask.pse__Parent_Task__c).add(pTask);
                            }
                            else
                            {
                                childProjectTasks.put(pTask.pse__Parent_Task__c, new list<pse__Project_Task__c>{pTask});    
                            }
                        }
                        else
                        {    
                            //Retrieve a map of field API names and values from current record
                            Map<String, Object> fieldstovalue = pTask.getPopulatedFieldsAsMap();
                            //CREATE NEW OBJECT FOR EACH RETRIVED PROJECT TASK RECORD
                            sObject sObj = Schema.getGlobalDescribe().get('pse__Project_Task__c').newSObject();
                            sObj.put('pse__Project__c', newlyCreatedProjId);
                            
                            //populating the required map with field labels as keys and their respective values as map values.
                            for(String fieldApiName : fieldstovalue.keySet())
                            {
                                Object fieldValue = fieldstovalue.get(fieldApiName);
                                if(fieldApiName != 'Id')
                                {
                                    if(fieldApiName != 'pse__Project__c' && fieldApiName != 'pse__Parent_Task__c' && fieldApiName != 'pse__Top_level_Parent_Task__c')
                                    {
                                        sObj.put(fieldApiName, fieldValue);    
                                    }
                                }
                            }
                            
                            if(sObj != null)
                            {
                                insertProjectTasks.add(sObj);    
                            }
                       // }    
                    }
                    
                    System.debug('childProjectTasks==>'+childProjectTasks);
                    System.debug('insertProjectTasks==>'+insertProjectTasks);
                    System.debug('Record Count==>'+insertProjectTasks.Size());
                    //INSERT LIST OF PROJECT TASKS RECORDS FOR NEWLY CREATED PROJECT
                    if(insertProjectTasks != null)
                    {
                        Set<Id> newPTIds = new Set<Id>();
                        Database.SaveResult[] srList = Database.insert(insertProjectTasks);
                        // Iterate through each returned result
                        for (Database.SaveResult sr : srList)
                        {
                            if(sr.isSuccess())
                            {
                                newPTIds.add(sr.getId());
                            }
                            else
                            {
                                // Operation failed, so get all errors                
                                for(Database.Error err : sr.getErrors()) 
                                {
                                    System.debug('The following error has occurred.');                    
                                    System.debug(err.getStatusCode() + ': ' + err.getMessage());
                                    System.debug('fields that affected this error: ' + err.getFields());
                                }
                            }
                        }

    //THIS METHOD IS USED TO GET PSA_Product_Package_Milestone__c RECORDS from MASTER PACKAGE
    Public Static Map<Id,List<PSA_Product_Package_Milestone__c>> getMilestonesforMasterPackage(Map<String,String> mapOfPackage)
    {
        Map<Id,List<PSA_Product_Package_Milestone__c>> mapOfPPM = new Map<Id,List<PSA_Product_Package_Milestone__c>>();
        Map<String,Map<String, Object>> mapofAttributeFieldandValue = new Map<String,Map<String, Object>>();
        Set<String> fieldsForQuery = new Set<String>();
        Map<String,set<PSA_Product_Package__c>> packageMapwithMaster = new Map<String,set<PSA_Product_Package__c>>();
        
        if(mapOfPackage != null)
        {
            //QUERY the CHILD PACKAGES FORM MASTER PACKAGES
            For(PSA_Product_Package__c mpp : [select id,Dimension_1_Field__c,Dimension_1_Value__c,Dimension_2_Field__c,Dimension_2_Value__c,
                                              Dimension_3_Field__c,Dimension_3_Value__c,Master_Product_Package__c from PSA_Product_Package__c 
                                              where Master_Product_Package__c IN: mapOfPackage.Values()])
            {
                //COLLECTING THE FIELDS FROM DIMENSION FIEDLS TO RETRIVE THE VALUES FROM ATTRIBUTE VALUE OBJECT 
                if(mpp.Dimension_1_Field__c != null)
                {
                    fieldsForQuery.add(mpp.Dimension_1_Field__c);    
                }
                
                if(mpp.Dimension_2_Field__c != null)
                {
                    fieldsForQuery.add(mpp.Dimension_2_Field__c);    
                }
                
                if(mpp.Dimension_3_Field__c != null)
                {
                    fieldsForQuery.add(mpp.Dimension_3_Field__c);    
                }
                
                //PREPARING THE MAP WITH MASTER & CHILD PACKAGE
                if(packageMapwithMaster != null & packageMapwithMaster.containskey(mpp.Master_Product_Package__c))
                {
                    packageMapwithMaster.get(mpp.Master_Product_Package__c).add(mpp);
                }
                else
                {
                    packageMapwithMaster.put(mpp.Master_Product_Package__c,new set<PSA_Product_Package__c>{mpp});
                }  
            }
            
            //QUERY THE Attribute Value Records
            String queryString = 'Select Id';
            for(String field : fieldsForQuery)
            {
                queryString += ', ' + field;
            }
            
            Set<String> attIds = mapOfPackage.Keyset();
            queryString += ' from Apttus_Config2__AssetAttributeValue__c where Id IN :attIds';
            
            for(Apttus_Config2__AssetAttributeValue__c attribute : Database.query(queryString))
            {
                mapofAttributeFieldandValue.put(attribute.Id,attribute.getPopulatedFieldsAsMap());
            }
            
            Map<Id,Id> mapofAttributeandpacIds = new Map<Id,Id>();
            //CHECK THE CONFIGURED DIMENSION VALUE WITH ATTRIBUTE VALUES CONFIGURED PARTICULAR FIELD
            for(String attributid : mapOfPackage.keyset())
            {
                if(mapofAttributeFieldandValue.containskey(attributid) 
                   && packageMapwithMaster.containskey(mapOfPackage.get(attributid)))
                {
                    for(PSA_Product_Package__c pack : packageMapwithMaster.get(mapOfPackage.get(attributid)))
                    {
                        //THIS IS CHECKING IF PRODUCT PACKAGE HAVEING 3 Dimension FIELD CONFIGURED
                        if(pack.Dimension_1_Field__c != null && pack.Dimension_2_Field__c != null && pack.Dimension_3_Field__c != null)
                        {
                            if(mapofAttributeFieldandValue.get(attributid).containskey(pack.Dimension_1_Field__c) && mapofAttributeFieldandValue.get(attributid).containskey(pack.Dimension_2_Field__c)
                              && mapofAttributeFieldandValue.get(attributid).containskey(pack.Dimension_3_Field__c))
                            {
                                if(String.Valueof(mapofAttributeFieldandValue.get(attributid).get(pack.Dimension_1_Field__c)) == String.valueof(pack.Dimension_1_Value__c)
                                  && String.Valueof(mapofAttributeFieldandValue.get(attributid).get(pack.Dimension_2_Field__c)) == String.valueof(pack.Dimension_2_Value__c)
                                  && String.valueof(mapofAttributeFieldandValue.get(attributid).get(pack.Dimension_3_Field__c)) == String.valueof(pack.Dimension_3_Value__c))
                                    mapofAttributeandpacIds.put(pack.Id,attributid);
                            }
                        }
                        else if(pack.Dimension_1_Field__c != null && pack.Dimension_2_Field__c != null)
                        {
                            if(mapofAttributeFieldandValue.get(attributid).containskey(pack.Dimension_1_Field__c) && mapofAttributeFieldandValue.get(attributid).containskey(pack.Dimension_2_Field__c))
                            {
